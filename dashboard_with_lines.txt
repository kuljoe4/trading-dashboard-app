   161	const GateBanner = ({ gateState, scannerPaused, reason, hibernating, activeTradesCount }) => {
   162	  if (!gateState && !scannerPaused) return null
   163
   164	  const messages = {
   165	    max_trades: 'Maximum open trades reached. Entry gated.',
   166	    max_trades_period: 'Maximum trades for the current period reached. Scanner paused.',

   168	    risk_pct: 'Total risk limit reached. Entries restricted.',
   169	    tod_risk: 'Historical performance risk for this hour. Entries blocked.',
   170	    sleeping: 'Engine idling outside trading windows.',
   171	    risk: 'Risk gate active. Monitoring only.',
   172	  }
   173
   174	  const isGatedIdle = (gateState === 'sleeping' || gateState === 'max_trades_period' || gateState === 'sl_guard') && activeTradesCount === 0;

   182	        scannerPaused ? "bg-red/10 border-red/20 text-red" : "bg-amber/10 border-amber/20 text-amber",

   184	      )}
   185	    >
   186	      <div className="flex items-center gap-3">
   187	        {hibernating ? <Zap size={16} className="text-accent animate-pulse opacity-50" /> : gateState === 'sleeping' ? <Pause size={16} className="animate-pulse" /> : <XCircle size={16} className={scannerPaused ? "animate-pulse" : ""} />}
   188	        <span className="uppercase tracking-widest">{messages[gateState] || 'Risk gate active.'}</span>
   189	        {hibernating ? (
   190	           <Tooltip content="Deep Sleep (Hibernation) Active: All market data connections closed to save maximum CPU/Memory. Engine will wake up automatically when limit expires.">
   191	            <div className="ml-auto bg-accent/20 px-2 py-0.5 rounded text-[10px] flex items-center gap-1.5 border border-accent/40 text-accent">
   192	              <Zap size={10} fill="currentColor" /> DEEP SLEEP
   193	            </div>
   194	          </Tooltip>
   195	        ) : isGatedIdle && (
   196	          <Tooltip content="Resource Suppression Active: Market feed and scanner are throttled to save CPU/Memory while idle.">
   197	            <div className="ml-auto bg-accent/10 px-2 py-0.5 rounded text-[10px] flex items-center gap-1.5 border border-accent/20">
   198	              <Leaf size={10} /> RESOURCE SAVER
   199	            </div>
   200	          </Tooltip>
   201	        )}
   202	      </div>
   203	      {reason && reason !== 'OK' && (
   204	        <div className="pl-7 opacity-80 font-mono text-[10px] tracking-tight">
   205	          Backend: {reason}
   206	        </div>
   207	      )}
   208	    </motion.div>
   209	  )

   212	const ScannerPreview = ({ scannerResults, config, onOpen }) => {

   214	  const top = scannerResults.slice(0, 5)

   226	        </div>
   227	        <button className="text-[11px] font-bold text-accent hover:text-accent/80 transition-colors uppercase tracking-widest" aria-label="View all scanner results" onClick={onOpen}>Open Full</button>
   228	      </div>
   229	      <div className="flex-1">
   230	        {top.length === 0 && placeholders.length === 5 ? (
   231	          <div className="h-full flex items-center justify-center text-dim text-[11px] font-bold uppercase tracking-widest bg-surface/10 animate-pulse">
   232	            Waiting for market data...
   233	          </div>
   234	        ) : (
   235	          <>
   236	            <AnimatePresence mode="popLayout">
   237	              {top.map((opp, i) => {
   238	                const passing = Math.abs(opp.pct) >= threshold
   239	                const isLong = opp.pct >= 0
   240	                const colorClass = isLong ? "text-green" : "text-red"
   241	                const isLast = i === top.length - 1 && placeholders.length === 0;
   242	                return (
   243	                  <motion.div
   244	                    key={opp.symbol}
   245	                    layout
   246	                    initial={{ opacity: 0 }}
   247	                    animate={{ opacity: 1 }}
   248	                    exit={{ opacity: 0 }}
   249	                    className={cn(
   250	                      "flex items-center gap-4 p-4 transition-colors hover:bg-white/5 h-[64px]",
   251	                      !isLast && "border-b border-border/40",
   252	                      !passing && "opacity-60"
   253	                    )}
   254	                  >
   255	                    <span className="text-[10px] text-dim font-mono w-4">#{i + 1}</span>
   256	                    <strong className="text-xs font-mono w-16">{opp.symbol.replace('USDT', '')}</strong>
   257	                    <div className="flex-1 flex justify-center h-8">
   258	                      <Sparkline data={opp.history} color={isLong ? "green" : "red"} width={48} height={20} />
   259	                    </div>
   260	                    <em className={cn("text-xs font-bold font-mono w-16 text-right", colorClass)}>
   261	                      {opp.pct >= 0 ? '+' : ''}{Number(opp.pct || 0).toFixed(2)}%
   262	                    </em>
   263	                    <b className={cn("text-[10px] font-bold w-12 text-right uppercase tracking-wider", passing ? "text-green" : "text-dim")}>
   264	                      {passing ? 'PASS' : 'WAIT'}
   265	                    </b>
   266	                  </motion.div>
   267	                )
   268	              })}
   269	            </AnimatePresence>
   270	            {placeholders.map((_, i) => (
   271	              <div key={`placeholder-${i}`} className={cn(
   272	                "h-[64px] flex items-center px-4 opacity-10 grayscale",
   273	                i !== placeholders.length - 1 && "border-b border-border/40"
   274	              )}>
   275	                <div className="w-4 h-2 bg-dim rounded-full mr-4" />
   276	                <div className="w-16 h-3 bg-dim rounded-full mr-4" />
   277	                <div className="flex-1" />
   278	                <div className="w-16 h-3 bg-dim rounded-full mr-4" />
   279	                <div className="w-12 h-2 bg-dim rounded-full" />
   280	              </div>
   281	            ))}
   282	          </>
   283	        )}
   284	      </div>
   285	    </div>
   286	  )
